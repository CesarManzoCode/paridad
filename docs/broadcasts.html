Cancel a broadcast

POST /broadcasts/{id}/cancel — destructiva

TypeScript

import { Resend } from "paridad-resend";

const resend = new Resend();
const { data, error } = await resend.broadcasts.cancel("49a3999c-0ce1-4ea6-ab68-afcd6dc2e794");
if (error !== null) {
  console.error(`${error.code}: ${error.message}`);
  process.exit(1);
}
console.log(JSON.stringify(data, null, 2));
