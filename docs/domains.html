domains/create

Create a new domain

POST /domains

TypeScript

import { Resend } from "paridad-resend";

// new Resend() reads RESEND_API_KEY and RESEND_BASE_URL.
const resend = new Resend();
const { data, error } = await resend.domains.create({
  "name": "example.com"
});
if (error !== null) {
  console.error(`${error.code}: ${error.message}`);
  process.exit(1);
}
console.log(JSON.stringify(data, null, 2));

package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"

	resend "github.com/CesarManzoCode/paridad/sdks/go"
)

func fail(err error) {
	fmt.Fprintln(os.Stderr, err)
	os.Exit(1)
}

func main() {
	// NewClient reads RESEND_API_KEY and RESEND_BASE_URL.
	c, err := resend.NewClient()
	if err != nil {
		fail(err)
	}
	ctx := context.Background()
	var body resend.CreateDomainRequest
	if err := json.Unmarshal([]byte(`{"name":"example.com"}`), &body); err != nil {
		fail(err)
	}
	result, err := c.Domains.Create(ctx, &body)
	if err != nil {
		fail(err)
	}
	out, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		fail(err)
	}
	fmt.Println(string(out))
}
