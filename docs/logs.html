logs/get

Retrieve a single log

GET /logs/{log_id}

TypeScript

import { Resend } from "paridad-resend";

// new Resend() reads RESEND_API_KEY and RESEND_BASE_URL.
const resend = new Resend();
const { data, error } = await resend.logs.get("49a3999c-0ce1-4ea6-ab68-afcd6dc2e794");
if (error !== null) {
  console.error(`${error.code}: ${error.message}`);
  process.exit(1);
}
console.log(JSON.stringify(data, null, 2));

Go

package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"

	resend "github.com/CesarManzoCode/paridad/sdks/go"
)

func fail(err error) {
	fmt.Fprintln(os.Stderr, err)
	os.Exit(1)
}

func main() {
	// NewClient reads RESEND_API_KEY and RESEND_BASE_URL.
	c, err := resend.NewClient()
	if err != nil {
		fail(err)
	}
	ctx := context.Background()
	result, err := c.Logs.Get(ctx, "49a3999c-0ce1-4ea6-ab68-afcd6dc2e794")
	if err != nil {
		fail(err)
	}
	out, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		fail(err)
	}
	fmt.Println(string(out))
}

CLI (bash)

#!/usr/bin/env bash
# paridad reads RESEND_API_KEY and RESEND_BASE_URL; it exits 1 on error.
set -euo pipefail
paridad logs get --log-id 49a3999c-0ce1-4ea6-ab68-afcd6dc2e794

Go

package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"

	resend "github.com/CesarManzoCode/paridad/sdks/go"
)

func fail(err error) {
	fmt.Fprintln(os.Stderr, err)
	os.Exit(1)
}

func main() {
	// NewClient reads RESEND_API_KEY and RESEND_BASE_URL.
	c, err := resend.NewClient()
	if err != nil {
		fail(err)
	}
	ctx := context.Background()
	var query resend.LogsListQuery
	if err := json.Unmarshal([]byte(`{}`), &query); err != nil {
		fail(err)
	}
	result, err := c.Logs.List(ctx, &query)
	if err != nil {
		fail(err)
	}
	out, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		fail(err)
	}
	fmt.Println(string(out))
}
